cmake_minimum_required(VERSION 3.22)
project(coredump_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LingmoToolkit REQUIRED)
find_package(GTest REQUIRED)
set(CRASH_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../lingmo-services/crash)
add_executable(coredump_test coredump_test.cpp ${CRASH_DIR}/coredump.cpp)
target_include_directories(coredump_test PRIVATE ${CRASH_DIR})
target_link_libraries(coredump_test PRIVATE Lingmo::ToolkitBase GTest::gtest_main)
enable_testing()
add_test(NAME coredump_test COMMAND coredump_test)

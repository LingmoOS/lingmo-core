cmake_minimum_required(VERSION 3.22)
project(sound_service_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LingmoSDK REQUIRED)
find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
add_executable(sound-service-test sound_test.cpp ../../lingmo-services/sound/soundservice.cpp)
target_include_directories(sound-service-test PRIVATE ../../lingmo-services)
target_link_libraries(sound-service-test PRIVATE Lingmo::SDKBase)
enable_testing()
add_test(NAME sound-service COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:sound-service-test>)
